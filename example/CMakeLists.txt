cmake_minimum_required(VERSION 3.24)
project(iiVisualCodecExample LANGUAGES CXX)
include(CTest)
find_package(iiVisualCodec 0.4 CONFIG REQUIRED COMPONENTS Image Spatial Video USD Vision OpenCV Geometry Asset3D CAD)
add_executable(visual_pipeline pipeline.cpp)
target_link_libraries(visual_pipeline PRIVATE iiVisualCodec::iiVisualCodec)
add_test(NAME iiVisualCodec.Example COMMAND visual_pipeline)
add_executable(vision_pipeline vision.cpp)
target_link_libraries(vision_pipeline PRIVATE iiVisualCodec::OpenCV)
add_executable(geometry_pipeline geometry.cpp)
target_link_libraries(geometry_pipeline PRIVATE iiVisualCodec::CAD iiVisualCodec::Asset3D iiVisualCodec::USD)
add_test(NAME iiVisualCodec.GeometryExample COMMAND geometry_pipeline)
