cmake_minimum_required(VERSION 3.24)
project(iiImageCodec VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(PNG 1.6 REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(WebP 1 CONFIG REQUIRED)
find_package(TIFF 4.5 REQUIRED)
add_library(iiImageCodec STATIC src/iiImageCodec/ImageCodec.cpp src/iiImageCodec/Jpeg.c
    src/iiImageCodec/Extended.cpp src/iiImageCodec/Metadata.cpp src/iiImageCodec/Thumbnail.cpp)
add_library(iiImageCodec::iiImageCodec ALIAS iiImageCodec)
if(APPLE)
    target_sources(iiImageCodec PRIVATE src/iiImageCodec/ColorMac.cpp)
    target_link_libraries(iiImageCodec PRIVATE "-framework CoreGraphics" "-framework CoreFoundation")
elseif(WIN32)
    target_sources(iiImageCodec PRIVATE src/iiImageCodec/ColorWindows.cpp)
    target_link_libraries(iiImageCodec PRIVATE windowscodecs ole32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(ColorManagement REQUIRED IMPORTED_TARGET lcms2)
    target_sources(iiImageCodec PRIVATE src/iiImageCodec/ColorLinux.cpp)
    target_link_libraries(iiImageCodec PRIVATE PkgConfig::ColorManagement)
else()
    target_sources(iiImageCodec PRIVATE src/iiImageCodec/ColorUnsupported.cpp)
endif()
target_compile_features(iiImageCodec PUBLIC cxx_std_23)
target_include_directories(iiImageCodec PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(iiImageCodec PRIVATE PNG::PNG JPEG::JPEG TIFF::TIFF ZLIB::ZLIB WebP::webp)
target_include_directories(iiImageCodec PRIVATE third_party/stb)
set_target_properties(iiImageCodec PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_executable(iiImageCodecThumbnailContracts tests/thumbnail.cpp)
    target_link_libraries(iiImageCodecThumbnailContracts PRIVATE iiImageCodec::iiImageCodec)
    add_test(NAME iiImageCodec.Thumbnail COMMAND iiImageCodecThumbnailContracts)
    add_executable(iiImageCodecColorContracts tests/color.cpp)
    target_link_libraries(iiImageCodecColorContracts PRIVATE iiImageCodec::iiImageCodec)
    add_test(NAME iiImageCodec.Color COMMAND iiImageCodecColorContracts)
    add_executable(iiImageCodecContracts tests/contracts.cpp)
    target_link_libraries(iiImageCodecContracts PRIVATE iiImageCodec::iiImageCodec)
    add_test(NAME iiImageCodec.Contracts COMMAND iiImageCodecContracts)
endif()
install(TARGETS iiImageCodec EXPORT iiImageCodecTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/iiImageCodec/ImageCodec.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiImageCodec)
install(EXPORT iiImageCodecTargets NAMESPACE iiImageCodec:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiImageCodec)
configure_package_config_file(cmake/iiImageCodecConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/iiImageCodecConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiImageCodec)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiImageCodecConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiImageCodecConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiImageCodecConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiImageCodec)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/iiImageCodec)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiImageCodec/docs)
install(DIRECTORY third_party/stb/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/iiImageCodec-stb
    FILES_MATCHING PATTERN "LICENSE*" PATTERN "README*")
