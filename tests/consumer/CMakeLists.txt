cmake_minimum_required(VERSION 3.24)
project(iiVisualCodecInstalledContracts LANGUAGES CXX)
include(CTest)
option(VISUAL_IMAGE_ONLY "Only import the image package and the legacy forwarding package" OFF)
option(VISUAL_VISION_ONLY "Import portable Vision objects without OpenCV, video or USD" OFF)
option(VISUAL_GEOMETRY_ONLY "Import portable Geometry without Assimp, CAD, OpenCV, video or USD" OFF)
option(VISUAL_CAD_ONLY "Import exact CAD without Assimp or other optional backends" OFF)
option(VISUAL_3D_ONLY "Import Assimp assets without CAD or other optional backends" OFF)
option(VISUAL_VIDEO_ONLY "Import native video without USD, OpenCV or 3D backends" OFF)
find_package(iiVisualCodec 0.2 CONFIG REQUIRED COMPONENTS Image)
find_package(iiImageCodec 0.1 CONFIG REQUIRED)
add_executable(legacy ../legacy.cpp)
target_link_libraries(legacy PRIVATE iiImageCodec::iiImageCodec)
add_test(NAME iiVisualCodec.InstalledLegacy COMMAND legacy)
if(VISUAL_CAD_ONLY OR VISUAL_3D_ONLY)
    if(VISUAL_CAD_ONLY)
        set(component CAD)
        set(contract cad)
    else()
        set(component Asset3D)
        set(contract asset3d)
    endif()
    find_package(iiVisualCodec 0.4 CONFIG REQUIRED COMPONENTS ${component})
    add_executable(geometry_backend ../${contract}.cpp)
    target_link_libraries(geometry_backend PRIVATE iiVisualCodec::${component})
    add_test(NAME iiVisualCodec.Installed${component} COMMAND geometry_backend)
    if(TARGET opencv_core OR TARGET PkgConfig::VisualFFmpeg OR TARGET usd)
        message(FATAL_ERROR "Geometry backend imported an unrelated media backend")
    endif()
    if((VISUAL_CAD_ONLY AND TARGET assimp::assimp) OR (VISUAL_3D_ONLY AND TARGET TKernel))
        message(FATAL_ERROR "3D and CAD backends are coupled")
    endif()
elseif(VISUAL_VIDEO_ONLY)
    find_package(iiVisualCodec 0.4.2 CONFIG REQUIRED COMPONENTS Video)
    add_executable(video_display ../video_display.cpp)
    target_link_libraries(video_display PRIVATE iiVisualCodec::Video)
    target_compile_definitions(video_display PRIVATE
        IIVISUALCODEC_VIDEO_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../resources/video")
    add_test(NAME iiVisualCodec.InstalledVideoDisplay COMMAND video_display)
    if(TARGET opencv_core OR TARGET usd OR TARGET assimp::assimp OR TARGET TKernel)
        message(FATAL_ERROR "Video-only discovery imported an unrelated backend")
    endif()
elseif(VISUAL_GEOMETRY_ONLY)
    find_package(iiVisualCodec 0.4 CONFIG REQUIRED COMPONENTS Geometry)
    add_executable(geometry ../geometry.cpp)
    target_link_libraries(geometry PRIVATE iiVisualCodec::Geometry)
    add_test(NAME iiVisualCodec.InstalledGeometry COMMAND geometry)
    if(TARGET assimp::assimp OR TARGET TKernel OR TARGET opencv_core OR TARGET PkgConfig::VisualFFmpeg OR TARGET usd)
        message(FATAL_ERROR "Portable Geometry imported an optional backend")
    endif()
elseif(VISUAL_VISION_ONLY)
    find_package(iiVisualCodec 0.3 CONFIG REQUIRED COMPONENTS Vision)
    add_executable(vision ../vision.cpp)
    target_link_libraries(vision PRIVATE iiVisualCodec::Vision)
    add_test(NAME iiVisualCodec.InstalledVision COMMAND vision)
    if(TARGET opencv_core OR TARGET PkgConfig::VisualFFmpeg OR TARGET usd OR TARGET assimp::assimp OR TARGET TKernel)
        message(FATAL_ERROR "Portable Vision objects imported an optional backend")
    endif()
elseif(NOT VISUAL_IMAGE_ONLY)
    find_package(iiVisualCodec 0.4 CONFIG REQUIRED COMPONENTS Spatial Video USD Vision OpenCV Geometry Asset3D CAD)
    add_executable(pipeline ../../example/pipeline.cpp)
    target_link_libraries(pipeline PRIVATE iiVisualCodec::iiVisualCodec)
    add_test(NAME iiVisualCodec.InstalledPipeline COMMAND pipeline)
    add_executable(opencv_contract ../opencv.cpp)
    target_link_libraries(opencv_contract PRIVATE iiVisualCodec::OpenCV)
    add_test(NAME iiVisualCodec.InstalledOpenCV COMMAND opencv_contract "${CMAKE_CURRENT_BINARY_DIR}/vision-contracts")
    add_executable(vision_pipeline ../../example/vision.cpp)
    target_link_libraries(vision_pipeline PRIVATE iiVisualCodec::OpenCV)
    add_test(NAME iiVisualCodec.InstalledVisionExample COMMAND vision_pipeline
        "${CMAKE_CURRENT_BINARY_DIR}/vision-contracts/red.png"
        "${CMAKE_CURRENT_BINARY_DIR}/vision-contracts/gray.png")
    set_tests_properties(iiVisualCodec.InstalledVisionExample PROPERTIES DEPENDS iiVisualCodec.InstalledOpenCV)
    foreach(component IN ITEMS geometry asset3d cad)
        if(component STREQUAL "geometry")
            set(imported Geometry)
        elseif(component STREQUAL "asset3d")
            set(imported Asset3D)
        else()
            set(imported CAD)
        endif()
        add_executable(${component} ../${component}.cpp)
        target_link_libraries(${component} PRIVATE iiVisualCodec::${imported})
        add_test(NAME iiVisualCodec.Installed.${component} COMMAND ${component})
    endforeach()
    add_executable(geometry_pipeline ../../example/geometry.cpp)
    target_link_libraries(geometry_pipeline PRIVATE iiVisualCodec::CAD iiVisualCodec::Asset3D iiVisualCodec::USD)
    add_test(NAME iiVisualCodec.InstalledGeometryExample COMMAND geometry_pipeline)
elseif(TARGET opencv_core OR TARGET PkgConfig::VisualFFmpeg OR TARGET usd OR TARGET assimp::assimp OR TARGET TKernel)
    message(FATAL_ERROR "Image-only discovery imported a video/USD backend")
endif()
if(TARGET Qt5::Core OR TARGET Qt6::Core)
    message(FATAL_ERROR "Visual codec package imported Qt")
endif()
